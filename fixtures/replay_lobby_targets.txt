## 4. 复跑命令（量上述数字）

```bash
# 单帧 → 每个 bbox（与写这份文件时用的是同一条命令）
python3 - <<'PY'
import numpy as np
from PIL import Image
BG = np.array([0x0e, 0x25, 0x1c])          # 底色 #0E251C（真帧第一主色）
a = np.asarray(Image.open('<W>/cap_orig/frames/replay_lobby_1.png').convert('RGB')).astype(np.int16)
non = np.abs(a - BG).max(axis=2) > 0
def bbox(m):
    ys, xs = np.nonzero(m); return (int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()))
for name, y0, y1, x0, x1 in [('date', 100, 166, 0, 700), ('level', 100, 166, 700, 1200)]:
    m = np.zeros_like(non); m[y0:y1+1, x0:x1+1] = non[y0:y1+1, x0:x1+1]; print(name, bbox(m))
PY
```
